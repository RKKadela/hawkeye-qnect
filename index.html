simulation_pipeline.py 
import qnect as qn

# Configure HawkEye QNeCT simulation network
net = qn.QuantumNetwork(name="HawkEye-Urban-Link")
alice = net.add_node("Alice", memory_type="87Rb")
bob = net.add_node("Bob", memory_type="87Rb")

# Link fiber parameters
link = net.connect_nodes(alice, bob, distance_km=45, loss_db_km=0.20)
link.apply_noise(qn.noise.DepolarizingChannel(gamma=0.03))

# Execute entanglement swapping & measure state
rho = net.run_swapping_protocol(fidelity_threshold=0.85)
print(f"Bell Fidelity: {rho.fidelity():.4f}")
